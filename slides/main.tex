% ---- Definition / equation block -------------------------------------------
\begin{frame}{A definition block}
  \begin{block}{CRPS --- Continuous Ranked Probability Score}
    How well the full forecast distribution matches the observed value:
    \[
      \mathrm{CRPS} = \int_{-\infty}^{\infty}\bigl[F(x) - H(x-y)\bigr]^2\,dx
    \]
    \begin{itemize}
      \item $F(x)$ forecast CDF; \; $H(x-y)$ step at the observation $y$
      \item Lower is better; reduces to $|f-y|$ for a deterministic forecast
    \end{itemize}
  \end{block}
\end{frame}
